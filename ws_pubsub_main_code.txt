"""
Redis Pub/Sub Publisher
Publishes events (messages, typing, read receipts) to chat:{chat_id} channels.
"""
import json
import uuid
from datetime import datetime
from redis.asyncio import Redis


class Publisher:
    def __init__(self, redis_client: Redis):
        self.redis = redis_client

    async def publish_message(
        self,
        chat_id: uuid.UUID,
        message_id: uuid.UUID,
        sender_id: uuid.UUID,
        content: str,
        sent_at: datetime,
    ):
        """Publish a new message to the chat channel."""
        payload = {
            "type": "message",
            "chat_id": str(chat_id),
            "message_id": str(message_id),
            "sender_id": str(sender_id),
            "content": content,
            "sent_at": sent_at.isoformat(),
        }
        channel = f"chat:{chat_id}"
        await self.redis.publish(channel, json.dumps(payload))

    async def publish_typing(self, chat_id: uuid.UUID, user_id: uuid.UUID):
        """Publish a typing indicator to the chat channel."""
        payload = {
            "type": "typing",
            "chat_id": str(chat_id),
            "user_id": str(user_id),
        }
        channel = f"chat:{chat_id}"
        await self.redis.publish(channel, json.dumps(payload))

    async def publish_read(
        self, chat_id: uuid.UUID, message_id: uuid.UUID, user_id: uuid.UUID
    ):
        """Publish a read receipt to the chat channel."""
        payload = {
            "type": "read",
            "chat_id": str(chat_id),
            "message_id": str(message_id),
            "user_id": str(user_id),
        }
        channel = f"chat:{chat_id}"
        await self.redis.publish(channel, json.dumps(payload))

"""
Redis Pub/Sub Subscriber
Background listener that receives messages from Redis channels
and forwards them to locally connected WebSocket clients.
"""
import json
import uuid
import asyncio
import logging
from redis.asyncio import Redis

logger = logging.getLogger(__name__)


class PubSubListener:
    """
    Listens to Redis Pub/Sub channels and delivers messages
    to the ConnectionManager for local WebSocket delivery.
    
    The `manager` is injected at init to avoid circular imports.
    """

    def __init__(self, redis_client: Redis, manager):
        self.pubsub = redis_client.pubsub()
        self.manager = manager  # websocket.manager.ConnectionManager
        self._running = False

    async def subscribe(self, chat_ids: list[uuid.UUID]):
        """Subscribe to one or more chat channels."""
        if not chat_ids:
            return
        channels = [f"chat:{cid}" for cid in chat_ids]
        await self.pubsub.subscribe(*channels)
        logger.info(f"Subscribed to {len(channels)} channel(s)")

    async def unsubscribe(self, chat_ids: list[uuid.UUID]):
        """Unsubscribe from one or more chat channels."""
        if not chat_ids:
            return
        channels = [f"chat:{cid}" for cid in chat_ids]
        await self.pubsub.unsubscribe(*channels)
        logger.info(f"Unsubscribed from {len(channels)} channel(s)")

    async def listen(self):
        """
        Main listener loop — runs as a background asyncio task.
        Receives messages from all subscribed channels and delivers
        them to the correct local WebSocket connections.
        """
        self._running = True
        logger.info("PubSub listener started")

        try:
            async for message in self.pubsub.listen():
                if not self._running:
                    break

                # Skip subscribe/unsubscribe confirmation messages
                if message["type"] not in ("message",):
                    continue

                try:
                    channel = message["channel"]
                    if isinstance(channel, bytes):
                        channel = channel.decode("utf-8")

                    data_raw = message["data"]
                    if isinstance(data_raw, bytes):
                        data_raw = data_raw.decode("utf-8")

                    data = json.loads(data_raw)

                    # Extract chat_id from channel name "chat:<uuid>"
                    chat_id_str = channel.split(":", 1)[1]
                    chat_id = uuid.UUID(chat_id_str)

                    # Get the sender so we can skip them (they already got an ack)
                    sender_id_str = data.get("sender_id") or data.get("user_id")
                    sender_id = uuid.UUID(sender_id_str) if sender_id_str else None

                    # Deliver to all local users in this chat EXCEPT the sender
                    await self.manager.deliver_to_chat(
                        chat_id=chat_id,
                        payload=data,
                        exclude_user_id=sender_id,
                    )

                except (json.JSONDecodeError, KeyError, ValueError) as e:
                    logger.error(f"Failed to process pubsub message: {e}")
                    continue

        except asyncio.CancelledError:
            logger.info("PubSub listener cancelled")
        except Exception as e:
            logger.error(f"PubSub listener error: {e}")
        finally:
            self._running = False
            await self.pubsub.close()
            logger.info("PubSub listener stopped")

    async def stop(self):
        """Gracefully stop the listener."""
        self._running = False
        await self.pubsub.close()

"""
WebSocket Message Handlers
Dispatches incoming WebSocket messages by their "type" field.
"""
import uuid
import logging
from sqlmodel.ext.asyncio.session import AsyncSession
from db.models import User
from chats.service import find_existing_chat, add_message_to_chat
from messages.service import read_message_byID
from pubsub.publisher import Publisher

logger = logging.getLogger(__name__)


async def handle_incoming(
    data: dict,
    user: User,
    session: AsyncSession,
    publisher: Publisher,
    manager,  # ConnectionManager — not typed to avoid circular import
):
    """
    Main dispatcher — routes incoming WebSocket JSON messages
    to the correct handler based on the "type" field.
    """
    msg_type = data.get("type")

    if msg_type == "message":
        await handle_message(data, user, session, publisher, manager)
    elif msg_type == "typing":
        await handle_typing(data, user, publisher)
    elif msg_type == "read":
        await handle_read(data, user, session, publisher)
    elif msg_type == "ping":
        await handle_ping(user, manager)
    else:
        await manager.send_to_user(user.id, {
            "type": "error",
            "detail": f"Unknown message type: {msg_type}",
        })


async def handle_message(
    data: dict,
    user: User,
    session: AsyncSession,
    publisher: Publisher,
    manager,
):
    """
    Handle a new chat message:
    1. Validate chat_id
    2. Persist to PostgreSQL (reuse existing chats/service)
    3. Send ack to sender
    4. Publish to Redis for delivery to recipient
    """
    chat_id_str = data.get("chat_id")
    content = data.get("content")

    if not chat_id_str or not content:
        await manager.send_to_user(user.id, {
            "type": "error",
            "detail": "Missing chat_id or content",
        })
        return

    try:
        chat_id = uuid.UUID(chat_id_str)
    except ValueError:
        await manager.send_to_user(user.id, {
            "type": "error",
            "detail": "Invalid chat_id format",
        })
        return

    # Persist the message using existing service
    try:
        message = await add_message_to_chat(
            session=session,
            chat_id=chat_id,
            sender_id=user.id,
            content=content,
        )
    except Exception as e:
        logger.error(f"Failed to persist message: {e}")
        await manager.send_to_user(user.id, {
            "type": "error",
            "detail": "Failed to save message",
        })
        return

    # Send ack to sender
    await manager.send_to_user(user.id, {
        "type": "message_ack",
        "message_id": str(message.id),
        "chat_id": str(chat_id),
        "sent_at": message.sent_at.isoformat(),
    })

    # Publish to Redis so the recipient gets it
    await publisher.publish_message(
        chat_id=chat_id,
        message_id=message.id,
        sender_id=user.id,
        content=content,
        sent_at=message.sent_at,
    )


async def handle_typing(data: dict, user: User, publisher: Publisher):
    """Publish a typing indicator to the chat channel."""
    chat_id_str = data.get("chat_id")
    if not chat_id_str:
        return

    try:
        chat_id = uuid.UUID(chat_id_str)
    except ValueError:
        return

    await publisher.publish_typing(chat_id=chat_id, user_id=user.id)


async def handle_read(
    data: dict,
    user: User,
    session: AsyncSession,
    publisher: Publisher,
):
    """
    Handle a read receipt:
    1. Update message status in DB (reuse existing messages/service)
    2. Publish to Redis so sender knows it was read
    """
    chat_id_str = data.get("chat_id")
    message_id_str = data.get("message_id")

    if not chat_id_str or not message_id_str:
        return

    try:
        chat_id = uuid.UUID(chat_id_str)
        message_id = uuid.UUID(message_id_str)
    except ValueError:
        return

    # Update in DB using existing service
    try:
        await read_message_byID(
            message_id=message_id,
            user_id=user.id,
            session=session,
        )
    except Exception as e:
        logger.error(f"Failed to update read receipt: {e}")
        return

    # Publish so the other user knows
    await publisher.publish_read(
        chat_id=chat_id,
        message_id=message_id,
        user_id=user.id,
    )


async def handle_ping(user: User, manager):
    """Respond to a ping with a pong and refresh presence."""
    await manager.refresh_presence(user.id)
    await manager.send_to_user(user.id, {"type": "pong"})


=== server\websocket\manager.py ===
"""
WebSocket Connection Manager
In-memory registry of user_id → WebSocket for this server instance.
Also handles presence tracking in Redis and Pub/Sub channel subscriptions.
"""
import uuid
import logging
from datetime import datetime
from fastapi.websockets import WebSocket
from redis.asyncio import Redis

logger = logging.getLogger(__name__)


class ConnectionManager:
    """
    Tracks which users are connected via WebSocket on THIS server.
    
    - connections:  user_id → WebSocket (single device per user)
    - user_chats:   user_id → list of chat_ids (cached on connect)
    - chat_members: chat_id → set of user_ids connected locally
    """

    def __init__(self, redis_client: Redis, subscriber):
        self.redis = redis_client
        self.subscriber = subscriber  # pubsub.PubSubListener (injected)
        self.connections: dict[uuid.UUID, WebSocket] = {}
        self.user_chats: dict[uuid.UUID, list[uuid.UUID]] = {}
        self.chat_members: dict[uuid.UUID, set[uuid.UUID]] = {}

    async def register(
        self, user_id: uuid.UUID, websocket: WebSocket, chat_ids: list[uuid.UUID]
    ):
        """Register a user's WebSocket connection."""
        # If user already connected, close old socket
        if user_id in self.connections:
            try:
                old_ws = self.connections[user_id]
                await old_ws.close(code=4001, reason="New connection opened")
            except Exception:
                pass

        self.connections[user_id] = websocket
        self.user_chats[user_id] = chat_ids

        # Track which users are in which chats (for local delivery)
        for chat_id in chat_ids:
            if chat_id not in self.chat_members:
                self.chat_members[chat_id] = set()
            self.chat_members[chat_id].add(user_id)

        # Set presence in Redis with TTL
        await self.redis.set(
            f"presence:{user_id}", "online", ex=60
        )

        # Subscribe to user's chat channels in Redis Pub/Sub
        await self.subscriber.subscribe(chat_ids)

        logger.info(f"User {user_id} connected ({len(chat_ids)} chats)")

    async def unregister(self, user_id: uuid.UUID):
        """Remove a user's WebSocket connection."""
        chat_ids = self.user_chats.pop(user_id, [])
        self.connections.pop(user_id, None)

        # Remove from chat_members tracking
        for chat_id in chat_ids:
            if chat_id in self.chat_members:
                self.chat_members[chat_id].discard(user_id)
                if not self.chat_members[chat_id]:
                    del self.chat_members[chat_id]

        # Remove presence
        await self.redis.delete(f"presence:{user_id}")

        # Unsubscribe from channels only if no other user needs them
        channels_still_needed = set()
        for uid, cids in self.user_chats.items():
            channels_still_needed.update(cids)

        channels_to_unsub = [cid for cid in chat_ids if cid not in channels_still_needed]
        if channels_to_unsub:
            await self.subscriber.unsubscribe(channels_to_unsub)

        logger.info(f"User {user_id} disconnected")

    async def refresh_presence(self, user_id: uuid.UUID):
        """Refresh the presence TTL (called on heartbeat/ping)."""
        await self.redis.set(f"presence:{user_id}", "online", ex=60)

    def get_socket(self, user_id: uuid.UUID) -> WebSocket | None:
        """Get the WebSocket for a connected user."""
        return self.connections.get(user_id)

    def is_online(self, user_id: uuid.UUID) -> bool:
        """Check if a user is connected locally."""
        return user_id in self.connections

    async def send_to_user(self, user_id: uuid.UUID, payload: dict):
        """Send a JSON payload to a specific user's WebSocket."""
        ws = self.connections.get(user_id)
        if ws:
            try:
                await ws.send_json(payload)
            except Exception as e:
                logger.error(f"Failed to send to {user_id}: {e}")
                await self.unregister(user_id)

    async def deliver_to_chat(
        self,
        chat_id: uuid.UUID,
        payload: dict,
        exclude_user_id: uuid.UUID | None = None,
    ):
        """
        Deliver a payload to all locally connected users in a chat.
        Skips `exclude_user_id` (typically the sender).
        """
        members = self.chat_members.get(chat_id, set())
        for uid in members:
            if uid == exclude_user_id:
                continue
            await self.send_to_user(uid, payload)

"""
WebSocket Route
Single endpoint: ws://host/ws?token=<JWT>
Handles authentication, connection lifecycle, and message loop.
"""
import logging
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Query
from utils import decode_token
from auth.service import User_Service
from db.main import get_session
from chats.service import get_user_chat_ids
from .handlers import handle_incoming

logger = logging.getLogger(__name__)

ws_router = APIRouter()


@ws_router.websocket("/ws")
async def websocket_endpoint(
    websocket: WebSocket,
    token: str = Query(...),
):
    """
    Main WebSocket endpoint.

    Flow:
    1. Validate JWT from query param
    2. Load user from DB
    3. Accept WebSocket & register in ConnectionManager
    4. Subscribe to user's chat channels via Redis Pub/Sub
    5. Enter message loop
    6. On disconnect → cleanup
    """
    # ── 1. Auth ──────────────────────────────────────
    token_data = decode_token(token)
    if not token_data:
        await websocket.close(code=4001, reason="Invalid token")
        return

    email = token_data.get("user", {}).get("email")
    if not email:
        await websocket.close(code=4001, reason="Invalid token payload")
        return

    # ── 2. Load user from DB ─────────────────────────
    async for session in get_session():
        user = await User_Service().get_user_by_email(email, session)
        if not user:
            await websocket.close(code=4001, reason="User not found")
            return

        if not user.is_verified:
            await websocket.close(code=4003, reason="Account not verified")
            return

        # ── 3. Get user's chat ids ───────────────────
        chat_ids = await get_user_chat_ids(session=session, user_id=user.id)

        # ── 4. Accept & register ─────────────────────
        await websocket.accept()

        # Access shared instances from app.state
        manager = websocket.app.state.manager
        publisher = websocket.app.state.publisher

        await manager.register(
            user_id=user.id,
            websocket=websocket,
            chat_ids=chat_ids,
        )

        # Send connected confirmation
        await websocket.send_json({
            "type": "connected",
            "user_id": str(user.id),
        })

        logger.info(f"WebSocket connected: {user.username} ({user.id})")

        # ── 5. Message loop ──────────────────────────
        try:
            while True:
                data = await websocket.receive_json()

                # Each message gets a fresh session for DB operations
                async for msg_session in get_session():
                    await handle_incoming(
                        data=data,
                        user=user,
                        session=msg_session,
                        publisher=publisher,
                        manager=manager,
                    )

        except WebSocketDisconnect:
            logger.info(f"WebSocket disconnected: {user.username}")
        except Exception as e:
            logger.error(f"WebSocket error for {user.username}: {e}")
        finally:
            # ── 6. Cleanup ───────────────────────────
            await manager.unregister(user.id)

import asyncio
from fastapi import FastAPI
from fastapi.templating import Jinja2Templates
from fastapi.requests import Request
from contextlib import asynccontextmanager

from db.main import init_db, close_db
from db.redis import PubSub_Redis
from pubsub.publisher import Publisher
from pubsub.subscriber import PubSubListener
from websocket.manager import ConnectionManager
from websocket.routes import ws_router
from auth.routes import auth_router
from users.routes import user_router
from chats.routes import chat_router
from messages.routes import msg_router
from mailserver.routes import mail_router
from middleware import custome_simple_middle
from errors import register_error_handlers


@asynccontextmanager
async def lifespan(app: FastAPI):
    # ── Startup ──────────────────────────────────
    await init_db()

    # Create shared instances (order matters: subscriber needs manager reference)
    publisher = Publisher(redis_client=PubSub_Redis)

    # Create subscriber first (without manager), then manager, then link them
    subscriber = PubSubListener(redis_client=PubSub_Redis, manager=None)
    manager = ConnectionManager(redis_client=PubSub_Redis, subscriber=subscriber)
    subscriber.manager = manager  # inject manager into subscriber

    # Store in app.state so WebSocket routes can access them
    app.state.manager = manager
    app.state.publisher = publisher
    app.state.subscriber = subscriber

    # Start the background Pub/Sub listener
    listener_task = asyncio.create_task(subscriber.listen())

    yield

    # ── Shutdown ─────────────────────────────────
    listener_task.cancel()
    try:
        await listener_task
    except asyncio.CancelledError:
        pass
    await subscriber.stop()
    await PubSub_Redis.close()
    await close_db()


app = FastAPI(lifespan=lifespan)

import os
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
templates = Jinja2Templates(directory=os.path.join(BASE_DIR, "templates"))

# Register middleware, error handlers, and routers
custome_simple_middle(app)
register_error_handlers(app)

# REST routers
app.include_router(auth_router, prefix="/auth", tags=["auth"])
app.include_router(user_router, prefix="/users", tags=["users"])
app.include_router(chat_router)
app.include_router(msg_router)
app.include_router(mail_router, prefix="/mail", tags=["mail"])

# WebSocket router
app.include_router(ws_router)


@app.get('/')
async def root(request: Request):
    return templates.TemplateResponse(
        'index.html',
        {"request": request}
    )
